$ psql -X -w -v ON_ERROR_STOP=1 -f - <<'SQL'
SET TIME ZONE 'America/Costa_Rica';
SELECT CASE WHEN v.precision_gnss_m IS NULL THEN 'QGIS' ELSE 'QField' END AS origen, v.finca_id, v.lote, v.codigo_upm AS upm, g.ugm, v.estado_visita, to_char(v.fecha_visita,'DD/MM/YYYY HH24:MI:SS') AS fecha_visita, round(v.precision_gnss_m::numeric,1) AS precision_gnss_m, v.editado_por, to_char(v.fecha_edicion,'DD/MM/YYYY HH24:MI:SS') AS fecha_edicion
FROM mgn.finca_visita v LEFT JOIN mgn.v_tab_geo_poligonos g ON g.id = v.finca_id
WHERE v.estado_visita <> 'pendiente' AND v.editado_por = 'edit_lote_a'
ORDER BY origen, v.finca_id;
SELECT v.finca_id, v.lote, v.codigo_upm AS upm, g.ugm, v.estado_visita, v.editado_por, to_char(v.fecha_edicion,'DD/MM/YYYY HH24:MI:SS') AS fecha_edicion FROM mgn.finca_visita v LEFT JOIN mgn.v_tab_geo_poligonos g ON g.id = v.finca_id WHERE v.finca_id = 258;
SELECT editado_por, estado_visita, count(*) FROM mgn.finca_visita WHERE estado_visita <> 'pendiente' GROUP BY 1,2 ORDER BY 1,2;
SELECT lote, count(*) FROM mgn.finca_visita GROUP BY 1 ORDER BY 1;
SQL
